## 🚀 Quick Commands

# Deploy to GCP (from your Mac)
export GCP_PROJECT_ID="your-project-id"
./deploy-to-gcp.sh

# On TDX machine: Run example
./run-in-tdx.sh

# Verify attestations (Rust tool - recommended)
cd verify-tdx && cargo run --release

# Or use bash script
./verify-attestation.sh

# Copy results back (from Mac)
gcloud compute scp --recurse \
  risc0-tdx-vm:~/hello-world/tdx-output ./ \
  --zone=us-central1-a
